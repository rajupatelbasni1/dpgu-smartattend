# 1. Base Python image (Lightweight)
FROM python:3.10-slim

# 2. Set environment variables
ENV PYTHONDONTWRITEBYTECODE 1
ENV PYTHONUNBUFFERED 1

# 3. System dependencies install karein (🔥 UPDATE: libgl1 naye Linux ke hisaab se)
RUN apt-get update && apt-get install -y \
    cmake \
    build-essential \
    libgl1 \
    libglib2.0-0 \
    && rm -rf /var/lib/apt/lists/*

# 4. Working directory set karein
WORKDIR /app

# 5. Requirements copy karein aur install karein
COPY requirements.txt /app/
RUN pip install --upgrade pip
RUN pip install cmake
RUN pip install --no-cache-dir -r requirements.txt

# 6. Apna bacha hua code copy karein
COPY . /app/

# 7. Cloud Run par chalane ke liye Gunicorn command (Port 8080)
CMD exec gunicorn --bind :$PORT --workers 1 --threads 8 --timeout 0 indiajobs.wsgi:application
